file.onchange = () => {
  const f = file.files[0];
  if (!f) return;
  const form = new FormData();
  form.append('file', f);
  const xhr = new XMLHttpRequest();
  xhr.open('POST', '/api/qaisypro');
  const token = new URLSearchParams(location.search).get('token');
  if (token) xhr.setRequestHeader('Authorization', 'Bearer ' + token);
  bar.style.display = 'block'; out.textContent = '';
  xhr.upload.onprogress = (ev) => {
    if (!ev.lengthComputable) return;
    bar.value = (ev.loaded / ev.total) * 100;
    txt.textContent = `Qaisypro: ${mb(ev.loaded)} MB / ${mb(ev.total)} MB`;
  };
  xhr.onload = () => {
    let j = {}; try { j = JSON.parse(xhr.responseText); } catch {}
    txt.textContent = j.success ? 'Qaisypro: تم بنجاح' : 'Qaisypro: فشل — ' + (j.error || xhr.status);
    out.textContent = JSON.stringify(j, null, 2);
  };
  xhr.onerror = () => { txt.textContent = 'Qaisypro: network error'; };
  xhr.send(form);
};
